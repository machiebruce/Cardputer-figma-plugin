### Multi-Screen App

**FREE VERSION:**
```
❌ Not supported
Requires manual coding of:
- Screen enum
- State machine

- Individual draw functions

```

**PRO VERSION:**
```cpp
#include <M5Cardputer.h>

// ============================================================================
// Multi-Screen Navigation System
// Generated by Cardputer Pro Plugin
// ============================================================================

// Screen IDs
enum Screen {
  SCREEN_HOME,
  SCREEN_MENU,
  SCREEN_SETTINGS
};

Screen currentScreen = SCREEN_HOME;

void drawHome() {
  M5Cardputer.Display.fillScreen(0x0000);
  // Home screen UI...
}

void drawMenu() {
  M5Cardputer.Display.fillScreen(0x0000);
  // Menu screen UI...
}

void drawSettings() {
  M5Cardputer.Display.fillScreen(0x0000);
  // Settings screen UI...
}

void drawCurrentScreen() {
  switch (currentScreen) {
    case SCREEN_HOME: drawHome(); break;
    case SCREEN_MENU: drawMenu(); break;
    case SCREEN_SETTINGS: drawSettings(); break;
  }
}

void setup() {
  M5Cardputer.begin();
  M5Cardputer.Display.setRotation(1);
  Serial.begin(115200);
  
  drawCurrentScreen();
}

void loop() {
  M5Cardputer.update();
  
  if (M5Cardputer.Keyboard.isChange()) {
    if (M5Cardputer.Keyboard.isPressed()) {
      Keyboard_Class::KeysState status = M5Cardputer.Keyboard.keysState();
      
      // Navigation keys
      if (status.fn && status.tab) {
        currentScreen = (Screen)((currentScreen + 1) % 3);
        drawCurrentScreen();
        Serial.println("Screen changed");
      }
    }
  }
  
  delay(10);
}
```
